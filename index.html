        function toggleSidebar() {

            const sidebar = document.getElementById('sidebar');

            sidebar.classList.toggle('translate-x-full');

        }

        function switchTab(tab) {

            document.getElementById('tab-colors').classList.toggle('hidden', tab !== 'colors');

            document.getElementById('tab-info').classList.toggle('hidden', tab !== 'info');

           

            document.getElementById('tab-colors-btn').className = tab === 'colors' ? 'flex-1 py-3 px-2 text-center border-b-2 border-emerald-600 text-emerald-700 bg-white' : 'flex-1 py-3 px-2 text-center border-b-2 border-transparent hover:text-slate-900';

            document.getElementById('tab-info-btn').className = tab === 'info' ? 'flex-1 py-3 px-2 text-center border-b-2 border-emerald-600 text-emerald-700 bg-white' : 'flex-1 py-3 px-2 text-center border-b-2 border-transparent hover:text-slate-900';

        }

        function updateThemeColors() {

            const primary = document.getElementById('color-primary').value;

            const secondary = document.getElementById('color-secondary').value;

           

            document.documentElement.style.setProperty('--primary-color', primary);

            document.documentElement.style.setProperty('--secondary-color', secondary);

        }

        function updateGeneralData() {

            const dept = document.getElementById('input-dept').value;

            const school = document.getElementById('input-school').value;

            const teacher = document.getElementById('input-teacher').value;

            document.querySelectorAll('.data-dept').forEach(el => el.textContent = dept);

            document.querySelectorAll('.data-school').forEach(el => el.textContent = school);

            document.querySelectorAll('.data-teacher').forEach(el => el.textContent = teacher);

        }

        // Export to PDF using html2pdf.js

        function exportToPDF() {

            const element = document.querySelector('.main-container main');

            const opt = {

                margin:       0,

                filename:     'سجل_شواهد_الأداء_الوظيفي.pdf',

                image:        { type: 'jpeg', quality: 0.98 },

                html2canvas:  { scale: 2, useCORS: true },

                jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }

            };

           

            // Show alert or loading if needed

            alert('جاري إعداد وتحميل ملف PDF، قد يستغرق ذلك بضع ثوانٍ...');

            html2pdf().from(element).set(opt).save();

        }

        // Export to PowerPoint using PptxGenJS

        function exportToPPTX() {

            let pptx = new PptxGenJS();

            pptx.layout = 'LAYOUT_4x3'; // أو LAYOUT_16x9

            let teacherName = document.getElementById('input-teacher').value;

            let schoolName = document.getElementById('input-school').value;

            // Slide 1: Cover

            let slide1 = pptx.addSlide();

            slide1.background = { color: "F3F7F0" };

            slide1.addText("سجل توثيق شواهد الأداء الوظيفي", { x: 1, y: 1.5, w: '80%', align: 'center', fontSize: 28, bold: true, color: "3B5E2B" });

            slide1.addText(`المعلم / المعلمة: ${teacherName}`, { x: 1, y: 3, w: '80%', align: 'center', fontSize: 18, color: "1F2937" });

            slide1.addText(`المدرسة: ${schoolName}`, { x: 1, y: 3.8, w: '80%', align: 'center', fontSize: 14, color: "64748B" });

            // Slide 2: Basmala

            let slide2 = pptx.addSlide();

            slide2.addText("بسم الله الرحمن الرحيم", { x: 1, y: 2.5, w: '80%', align: 'center', fontSize: 32, bold: true, color: "3B5E2B" });

            pptx.writeFile({ fileName: "سجل_الشواهد.pptx" });

        }
